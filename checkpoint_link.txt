Loading:
    from transformers import AutoModelForVision2Seq
    from peft import PeftModel

    base = AutoModelForVision2Seq.from_pretrained(
        "openvla/openvla-7b", torch_dtype=torch.bfloat16,
        low_cpu_mem_usage=True, trust_remote_code=True,
    ).cuda()
    model = PeftModel.from_pretrained(base, "path/to/adapter")

Normalization:
    Base 使用预训练 config.json 中的 norm_stats["bridge_orig"]；
    Final 使用 zip 内的 dataset_statistics.json（{"bridge_subset": {...}}），
    推理时覆盖 model.base_model.model.norm_stats["bridge_orig"]。

Inference preprocessing:
    prompt = "In: What action should the robot take to {instruction}?\nOut:"
    observation: 224x224 RGB, Gaussian blur sigma=1.5
    action: 7-DoF [x,y,z,roll,pitch,yaw,gripper], gripper 1=open/0=close
